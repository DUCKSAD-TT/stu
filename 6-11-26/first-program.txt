int age = 5;
float GPAX = 3.74;
char blood_type = 'A';
String name = "Gunjapong";
int arr[10] = {2, 3, 8, 4, 6, 4, 9, 8, 7, 0};

void setup()
{
  Serial.begin(9600);
}

void loop()
{
  delay(1000);
  Serial.print("Name = ");
  Serial.println(name);
  Serial.print("age = ");
  Serial.println(age);
  Serial.print("GPAX = ");
  Serial.println(GPAX);
  Serial.print("Blood_type = ");
  Serial.println(blood_type);
  Serial.print("arr = ");
  for (int i = 0; i < sizeof(arr) / sizeof(arr[0]); i++) {
    Serial.print(arr[i]);
    Serial.print(" ");
    delay(100);
  }
}
