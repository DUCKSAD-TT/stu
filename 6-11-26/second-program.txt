#include <math.h>
int x = 5, y = 2, m, b, t, r, p;
float d, s;
char A[8] = {'m', 'b', 't', 'd', 'r', 'p', 's'};


void setup()
{
  Serial.begin(9600);
}

void loop()
{
  m = x + y;
  b = x - y;
  t = x * y;
  r = x % y;
  p = pow(x, 2) + 1;
  d = (float)x / y;
  s = sqrt(x);
  Serial.println(m);
  Serial.println(b);
  Serial.println(t);
  Serial.println(d);
  Serial.println(r);
  Serial.println(p);
  Serial.println(s);
  delay(1000);
}
